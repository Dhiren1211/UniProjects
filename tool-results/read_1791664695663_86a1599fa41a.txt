    79→function CountUp({ value, className }: { value: number; className?: string }) {
    80→  const ref = useRef<HTMLSpanElement>(null);
    81→  const inView = useInView(ref, { once: true, margin: "-40px" });
    82→
    83→  useEffect(() => {
    84→    if (!inView || !ref.current) return;
    85→    const controls = animate(0, value, {
    86→      duration: 1.4,
    87→      ease: "easeOut",
    88→      onUpdate: (v) => {
    89→        if (ref.current) ref.current.textContent = Math.round(v).toLocaleString();
    90→      },
    91→    });
    92→    return () => controls.stop();
    93→  }, [inView, value]);
    94→
    95→  return (
    96→    <span ref={ref} className={className}>
    97→      0
    98→    </span>
    99→  );
   100→}

   102→function RotatingWord() {
   103→  const [idx, setIdx] = useState(0);
   104→

   106→    const t = setInterval(() => setIdx((i) => (i + 1) % HEADLINE_WORDS.length), 2600);

   110→  return (
   111→    <span className="relative inline-grid align-bottom">

   118→          initial={false}
   119→          animate={
   120→            i === idx
   121→              ? { opacity: 1, y: 0, filter: "blur(0px)" }

   156→function HeroSearch({ navigate }: { navigate: Navigate }) {
   157→  const [q, setQ] = useState("");
   158→  const [results, setResults] = useState<SearchResults | null>(null);
   159→  const [open, setOpen] = useState(false);
   160→  const [loading, setLoading] = useState(false);
   161→  const boxRef = useRef<HTMLDivElement>(null);
   162→  const seqRef = useRef(0);
   163→
   164→  useEffect(() => {
   165→    const query = q.trim();
   166→    const t = setTimeout(() => {
   167→      if (query.length < 2) {
   168→        setResults(null);
   169→        setOpen(false);
   170→        setLoading(false);
   171→        return;
   172→      }
   173→      setLoading(true);
   174→      const seq = ++seqRef.current;
   175→      globalSearch(query)
   176→        .then((r) => {
   177→          if (seq !== seqRef.current) return;
   178→          setResults(r);
   179→          setOpen(true);
   180→        })
   181→        .catch(() => {})
   182→        .finally(() => {
   183→          if (seq === seqRef.current) setLoading(false);
   184→        });
   185→    }, 250);
   186→    return () => clearTimeout(t);
   187→  }, [q]);
   188→
   189→  useEffect(() => {
   190→    if (!open) return;
   191→    const onDocDown = (e: MouseEvent) => {
   192→      if (boxRef.current && !boxRef.current.contains(e.target as Node)) setOpen(false);
   193→    };
   194→    document.addEventListener("mousedown", onDocDown);
   195→    return () => document.removeEventListener("mousedown", onDocDown);
   196→  }, [open]);
   197→
   198→  const submit = () => {
   199→    setOpen(false);
   200→    navigate({ name: "projects", q: q.trim() || undefined });
   201→  };
   202→
   203→  const hasAny = results && (results.projects.length > 0 || results.users.length > 0);
   204→
   205→  return (
   206→    <div ref={boxRef} className="relative w-full max-w-xl">
   207→      {/* Bar */}

   210→        <input
   211→          value={q}
   212→          onChange={(e) => setQ(e.target.value)}
   213→          onFocus={() => {
   214→            if (results && q.trim().length >= 2) setOpen(true);
   215→          }}
   216→          onKeyDown={(e) => {
   217→            if (e.key === "Enter") submit();
   218→            if (e.key === "Escape") setOpen(false);
   219→          }}
   220→          placeholder="Search projects, people, tags…"
   221→          aria-label="Search the showcase"
   222→          className="h-9 min-w-0 flex-1 bg-transparent text-[15px] text-white outline-none placeholder:text-teal-100/50"

   224→        {loading ? <Loader2 className="size-4 shrink-0 animate-spin text-teal-200/70" aria-hidden /> : null}

   226→          onClick={submit}
   227→          className="h-9 rounded-xl bg-teal-400 px-4 text-sm font-semibold text-teal-950 shadow-none hover:bg-teal-300"
   228→        >
   229→          Search
   230→        </Button>
   231→      </div>

   233→      {/* Dropdown */}
   234→      {open ? (
   235→        <div className="absolute inset-x-0 top-full z-30 mt-2 overflow-hidden rounded-2xl border border-up-border bg-popover text-popover-foreground shadow-float">

   379→function FloatingCollage({
   380→  projects,
   381→  sx,
   382→  sy,
   383→  navigate,
   384→}: {

   386→  sx: ReturnType<typeof useSpring>;
   387→  sy: ReturnType<typeof useSpring>;

   498→      onMouseLeave={() => {

  1110→export function HomeView({ navigate }: { navigate: Navigate }) {
  1111→  const [stats, setStats] = useState<StatsDTO | null>(null);
  1112→  const [featured, setFeatured] = useState<ProjectListItem[] | null>(null);
  1113→  const [recent, setRecent] = useState<ProjectListItem[] | null>(null);
  1114→  const [departments, setDepartments] = useState<DepartmentDTO[] | null>(null);
  1115→  const [events, setEvents] = useState<EventDTO[] | null>(null);
  1116→
  1117→  useEffect(() => {
  1118→    const ac = new AbortController();
  1119→    fetchStats(ac.signal)
  1120→      .then(setStats)
  1121→      .catch(() => {});
  1122→    fetchDepartments(ac.signal)
  1123→      .then(setDepartments)
  1124→      .catch(() => {});
  1125→    fetchEvents(ac.signal)
  1126→      .then(setEvents)
  1127→      .catch(() => {});
  1128→    fetchProjects({ featured: "1", limit: 5 }, ac.signal)
  1129→      .then(setFeatured)
  1130→      .catch(() => {});
  1131→    fetchProjects({ sort: "newest", limit: 6 }, ac.signal)
  1132→      .then(setRecent)
  1133→      .catch(() => {});
  1134→    return () => ac.abort();
  1135→  }, []);
  1136→
  1137→  return (
  1138→    <div>
  1139→      <Hero navigate={navigate} stats={stats} recent={recent} />
  1140→      <Marquee projects={recent} />
  1141→      <StatsBand stats={stats} />
  1142→      <Spotlight navigate={navigate} projects={featured} />
  1143→      <Departments navigate={navigate} departments={departments} />
  1144→      <Events events={events} />
  1145→      <Latest navigate={navigate} recent={recent} />
  1146→      <CtaPanel navigate={navigate} />
  1147→    </div>
  1148→  );
  1149→}
