  3050→/** Memorable 12-char password guaranteed to mix upper, lower and digits. */
  3051→function generatePassword(): string {
  3052→  const lower = "abcdefghijkmnpqrstuvwxyz";
  3053→  const upper = "ABCDEFGHJKLMNPQRSTUVWXYZ";
  3054→  const digits = "23456789";
  3055→  const all = lower + upper + digits;
  3056→  const pick = (set: string) => set[Math.floor(Math.random() * set.length)];
  3057→  const out = [pick(upper), pick(lower), pick(digits)];
  3058→  for (let i = 0; i < 9; i++) out.push(pick(all));
  3059→  for (let i = out.length - 1; i > 0; i--) {
  3060→    const j = Math.floor(Math.random() * (i + 1));
  3061→    [out[i], out[j]] = [out[j], out[i]];
  3062→  }
  3063→  return out.join("");
  3064→}
