    61→function FadeIn({ children, delay = 0, className }: { children: ReactNode; delay?: number; className?: string }) {
    62→  const reduce = useReducedMotion();
    63→  return (
    64→    <motion.div
    65→      initial={reduce ? { opacity: 0 } : { opacity: 0, y: 14 }}
    66→      animate={reduce ? { opacity: 1 } : { opacity: 1, y: 0 }}
    67→      transition={{ duration: 0.4, delay, ease: "easeOut" }}
    68→      className={className}
    69→    >
    70→      {children}
    71→    </motion.div>
    72→  );
    73→}

    75→function labelForCapstone(capstone: string | null): string | null {

    78→  return capstone;

   207→function StudentDashboard({ user, navigate }: { user: SafeUserDTO; navigate: Navigate }) {
   208→  const isLeader = user.isTeamLeader;
   209→  const [project, setProject] = useState<ProjectListItem | null | undefined>(undefined);
   210→  // pending join-request counts per project slug — null means "not known yet"
   211→  const [joinsBySlug, setJoinsBySlug] = useState<Record<string, number>>({});
   212→
   213→  // my most relevant project (owned + memberships), newest first
   214→  useEffect(() => {
   215→    let cancelled = false;
   216→    fetchProjects({ status: "MINE", limit: 1 })
   217→      .then((rows) => {
   218→        if (!cancelled) setProject(rows[0] ?? null);
   219→      })
   220→      .catch(() => {
   221→        if (!cancelled) setProject(null);
   222→      });
   223→    return () => {
   224→      cancelled = true;
   225→    };
   226→  }, []);
   227→
   228→  // team leaders also watch incoming join requests for their project
   229→  const slug = project?.slug;
   230→  useEffect(() => {
   231→    if (!isLeader || !slug) return;
   232→    let cancelled = false;
   233→    fetchJoinRequests(slug)
   234→      .then((rows) => {
   235→        if (!cancelled)
   236→          setJoinsBySlug((prev) => ({
   237→            ...prev,
   238→            [slug]: rows.filter((r) => r.status === "PENDING").length,
   239→          }));
   240→      })
   241→      .catch(() => {
   242→        if (!cancelled) setJoinsBySlug((prev) => ({ ...prev, [slug]: 0 }));
   243→      });
   244→    return () => {
   245→      cancelled = true;
   246→    };
   247→  }, [isLeader, slug]);
   248→
   249→  const pendingJoins = slug ? (joinsBySlug[slug] ?? null) : null;
   250→
   251→  return (
   252→    <div className="grid gap-4 lg:grid-cols-[minmax(0,1fr)_20rem]">
   253→      <FadeIn className="min-w-0 space-y-4" delay={0.05}>
   254→        {/* ---------- my team ---------- */}
   255→        <section aria-label="My team" className="rounded-2xl border border-up-border bg-card p-5 shadow-card">
   256→          <h2 className="text-sm font-bold tracking-tight text-foreground">My team</h2>
   257→          <p className="mt-0.5 text-xs text-muted-foreground">Your capstone project and its status</p>
   258→
   259→          {project === undefined ? (
   260→            <div className="mt-4 space-y-2">
   261→              <Skeleton className="h-16 w-full rounded-xl" />
   262→              <Skeleton className="h-9 w-2/3 rounded-xl" />
   263→            </div>
   264→          ) : project === null ? (
   265→            isLeader ? (
   266→              <div className="mt-4 rounded-2xl border border-dashed border-up-border-strong bg-surface-muted/40 p-6 text-center">
   267→                <span className="mx-auto flex size-11 items-center justify-center rounded-2xl bg-primary-50 text-primary-700">
   268→                  <SquarePlus className="size-5" aria-hidden />
   269→                </span>
   270→                <p className="mt-3 font-semibold text-foreground">No project yet</p>
   271→                <p className="mx-auto mt-1 max-w-sm text-sm text-muted-foreground">
   272→                  You lead your team — submit your capstone project when it is ready for the showcase.
   273→                </p>
   274→                <Button
   275→                  onClick={() => navigate({ name: "create" })}
   276→                  className="mt-4 gap-2 rounded-xl px-5 font-semibold"
   277→                >
   278→                  <SquarePlus className="size-4" aria-hidden />
   279→                  Submit your project
   280→                </Button>
   281→              </div>
   282→            ) : (
   283→              <div className="mt-4 rounded-2xl border border-dashed border-up-border-strong bg-surface-muted/40 p-6 text-center">
   284→                <span className="mx-auto flex size-11 items-center justify-center rounded-2xl bg-surface-muted text-muted-foreground">
   285→                  <UsersRound className="size-5" aria-hidden />
   286→                </span>
   287→                <p className="mt-3 font-semibold text-foreground">Waiting on your team leader</p>
   288→                <p className="mx-auto mt-1 max-w-sm text-sm text-muted-foreground">
   289→                  Your team leader submits the project — ask them to publish it so your team shows up in
   290→                  the showcase.
   291→                </p>
   292→              </div>
   293→            )
   294→          ) : (
   295→            <div className="mt-4">
   296→              <article className="flex flex-wrap items-center gap-x-4 gap-y-3 rounded-xl border border-up-border bg-surface-muted/30 p-3">
   297→                {project.coverImage ? (
   298→                  <img
   299→                    src={project.coverImage}
   300→                    alt=""
   301→                    loading="lazy"
   302→                    className="size-14 shrink-0 rounded-lg object-cover"
   303→                  />
   304→                ) : (
   305→                  <span
   306→                    aria-hidden
   307→                    className="flex size-14 shrink-0 items-center justify-center rounded-lg bg-primary-50 text-primary-700"
   308→                  >
   309→                    <FolderKanban className="size-5" />
   310→                  </span>
   311→                )}
   312→                <div className="min-w-0 flex-1 basis-48">
   313→                  <button
   314→                    type="button"
   315→                    onClick={() => navigate({ name: "project", slug: project.slug })}
   316→                    className="truncate text-sm font-semibold text-foreground transition-colors hover:text-primary-700"
   317→                  >
   318→                    {project.title}
   319→                  </button>
   320→                  <div className="mt-1 flex flex-wrap items-center gap-1.5">
   321→                    <StatusBadge status={project.status} />
   322→                    {project.verified ? <VerifiedBadge withLabel /> : null}
   323→                    <span className="inline-flex items-center gap-1 text-[11px] text-muted-foreground">
   324→                      <Eye className="size-3" aria-hidden />
   325→                      {formatViews(project.views)}
   326→                    </span>
   327→                    <span className="inline-flex items-center gap-1 text-[11px] text-muted-foreground">
   328→                      <Users className="size-3" aria-hidden />
   329→                      {project.members.length} member{project.members.length === 1 ? "" : "s"}
   330→                    </span>
   331→                  </div>
   332→                  {project.supervisor ? (
   333→                    <p className="mt-1 text-[11px] text-muted-foreground">Supervisor · {project.supervisor}</p>
   334→                  ) : null}
   335→                </div>
   336→                <div className="flex shrink-0 items-center gap-2">
   337→                  {isLeader ? (
   338→                    <Button
   339→                      variant="outline"
   340→                      size="sm"
   341→                      className="h-9 rounded-xl px-3 text-xs font-semibold"
   342→                      onClick={() => navigate({ name: "create", editSlug: project.slug })}
   343→                    >
   344→                      Edit project
   345→                    </Button>
   346→                  ) : null}
   347→                  <Button
   348→                    size="sm"
   349→                    className="h-9 rounded-xl px-3 text-xs font-semibold"
   350→                    onClick={() => navigate({ name: "project", slug: project.slug })}
   351→                  >
   352→                    View page
   353→                  </Button>
   354→                </div>
   355→              </article>
   356→
   357→              {isLeader && pendingJoins !== null && pendingJoins > 0 ? (
   358→                <button
   359→                  type="button"
   360→                  onClick={() => navigate({ name: "project", slug: project.slug })}
   361→                  className="mt-2 flex w-full items-center gap-2 rounded-xl bg-primary-50 px-3 py-2 text-left text-xs font-semibold text-primary-700 transition-colors hover:bg-primary-100"
   362→                >
   363→                  <UserPlus className="size-4 shrink-0" aria-hidden />
   364→                  {pendingJoins} pending join request{pendingJoins === 1 ? "" : "s"} — review on the project
   365→                  page
   366→                </button>
   367→              ) : null}
   368→            </div>
   369→          )}
   370→        </section>
   371→      </FadeIn>
   372→
   373→      <FadeIn className="min-w-0 space-y-4" delay={0.1}>
   374→        <IdentityCard user={user} />
   375→        <SecurityRow navigate={navigate} />
   376→      </FadeIn>
   377→    </div>
   378→  );
   379→}

   463→function AdminDashboard({ user, navigate }: { user: SafeUserDTO; navigate: Navigate }) {

   509→          icon: GraduationCap,

   514→          label: "Projects",

   516→          icon: FolderKanban,
   517→          go: () => goAdmin("projects"),
   518→          tone: "bg-primary-50 text-primary-700",

   523→  const quickActions: { key: keyof typeof ADMIN_ACTION_ICONS; label: string; hint: string; tab: string }[] = [

   529→

   532→      <FadeIn className="min-w-0 space-y-4" delay={0.05}>

   534→        {failed ? (

   536→            icon={ShieldCheck}

   538→            hint="The admin overview API did not respond — try the admin console directly."
   539→            action={
   540→              <Button onClick={() => goAdmin()} className="rounded-xl font-semibold">
   541→                Open admin panel
   542→              </Button>
   543→            }
   544→          />
   545→        ) : (
   546→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 xl:grid-cols-5">
   547→            {tiles.map((t) => {

   551→                  key={t.label}
   552→                  type="button"
   553→                  onClick={t.go}
   554→                  className="group rounded-2xl border border-up-border bg-card p-4 text-left shadow-card outline-none transition-colors hover:border-up-border-strong focus-visible:ring-2 focus-visible:ring-ring"

   557→                    <Icon className="size-4" aria-hidden />
   558→                  </span>
   559→                  <p className="mt-2.5 text-2xl font-bold tabular-nums tracking-tight text-foreground">
   560→                    {overview ? t.value.toLocaleString() : <Skeleton className="h-7 w-12" />}
   561→                  </p>
   562→                  <p className="mt-0.5 text-xs font-semibold text-foreground">{t.label}</p>
   563→                </button>
   564→              );
   565→            })}
   566→          </div>
   567→        )}
   568→
   569→        {/* quick actions */}
   570→        <section aria-label="Quick actions" className="rounded-2xl border border-up-border bg-card p-5 shadow-card">
   571→          <h2 className="text-sm font-bold tracking-tight text-foreground">Quick actions</h2>

   577→                <button
   578→                  key={a.key}
   579→                  type="button"
   580→                  onClick={() => goAdmin(a.tab)}

   582→                >
   583→                  <span className="flex size-8 shrink-0 items-center justify-center rounded-xl bg-card text-primary-700 shadow-sm ring-1 ring-up-border">
   584→                    <Icon className="size-4" aria-hidden />
   585→                  </span>
   586→                  <span className="min-w-0 flex-1">

   588→                      {a.label}
   589→                    </span>
   590→                    <span className="block truncate text-xs text-muted-foreground">{a.hint}</span>
   591→                  </span>
   592→                </button>
   593→              );
   594→            })}
   595→          </div>
   596→        </section>
   597→
   598→        {/* recent activity */}
   599→        <section aria-label="Recent activity" className="rounded-2xl border border-up-border bg-card p-5 shadow-card">
   600→          <div className="flex items-center justify-between gap-3">

   603→              type="button"
   604→              onClick={() => goAdmin()}

   606→            >
   607→              Open admin panel
   608→            </button>
   609→          </div>
   610→          {overview === null ? (
   611→            <div className="mt-3 space-y-2">
   612→              {Array.from({ length: 3 }).map((_, i) => (
   613→                <Skeleton key={i} className="h-10 w-full rounded-xl" />
   614→              ))}
   615→            </div>
   616→          ) : overview.activity.length === 0 ? (
   617→            <p className="mt-3 text-sm text-muted-foreground">
   618→              Nothing has happened yet — activity lands here as members work.
   619→            </p>
   620→          ) : (
   621→            <ul className="mt-2 divide-y divide-up-border">
   622→              {overview.activity.slice(0, 5).map((item) => (
   623→                <li key={item.id} className="flex items-center gap-3 py-2">
   624→                  {(() => {
   625→                    const Icon = ACTIVITY_ICONS[item.kind] ?? FolderKanban;
   626→                    return (
   627→                      <span className="flex size-8 shrink-0 items-center justify-center rounded-xl bg-primary-50 text-primary-700">
   628→                        <Icon className="size-4" aria-hidden />
   629→                      </span>
   630→                    );
   631→                  })()}
   632→                  <span className="min-w-0 flex-1">
   633→                    <span className="block truncate text-sm font-medium text-foreground">{item.text}</span>
   634→                    {item.detail ? (
   635→                      <span className="block truncate text-xs text-muted-foreground">{item.detail}</span>
   636→                    ) : null}
   637→                  </span>
   638→                  <span className="shrink-0 whitespace-nowrap text-[11px] font-medium text-muted-foreground">
   639→                    {timeAgo(item.createdAt)}
   640→                  </span>
   641→                </li>
   642→              ))}
   643→            </ul>
   644→          )}
   645→        </section>
   646→      </FadeIn>
   647→
   648→      <FadeIn className="min-w-0 space-y-4" delay={0.1}>
   649→        <IdentityCard user={user} />
   650→        <section aria-label="Console shortcut" className="rounded-2xl border border-up-border bg-night p-5 text-white shadow-card">
   651→          <span className="flex size-9 items-center justify-center rounded-xl bg-white/10">
   652→            <ShieldCheck className="size-4.5" aria-hidden />
   653→          </span>
   654→          <p className="mt-3 text-sm font-bold">Full console access</p>
   655→          <p className="mt-1 text-xs leading-relaxed text-white/60">
   656→            Approvals, users, projects, announcements, moderation, emails and site settings.
   657→          </p>
   658→          <Button
   659→            onClick={() => goAdmin()}
   660→            variant="outline"
   661→            className="mt-4 w-full rounded-xl border-white/15 bg-white/10 font-semibold text-white hover:bg-white/20 hover:text-white"
   662→          >
   663→            Open admin panel
   664→          </Button>
   665→        </section>
   666→      </FadeIn>
   667→    </div>
   668→  );
   669→}
