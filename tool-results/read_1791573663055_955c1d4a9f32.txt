   123→function loadImageEl(src: string): Promise<HTMLImageElement> {
   124→  return new Promise((resolve, reject) => {
   125→    const img = new Image();
   126→    img.onload = () => resolve(img);
   127→    img.onerror = () => reject(new Error("Couldn't load that image"));
   128→    img.src = src;
   129→  });
   130→}

   132→/** Pixel-crop an image onto a canvas and return an upload-ready JPEG File. */
   133→async function cropToJpegFile(src: string, area: CropArea): Promise<File> {
   134→  const img = await loadImageEl(src);
   135→  const scale = Math.min(1, COVER_MAX_EDGE / Math.max(area.width, 1));
   136→  const canvas = document.createElement("canvas");
   137→  canvas.width = Math.max(1, Math.round(area.width * scale));
   138→  canvas.height = Math.max(1, Math.round(area.height * scale));
   139→  const ctx = canvas.getContext("2d");
   140→  if (!ctx) throw new Error("Your browser can't process images right now");
   141→  ctx.drawImage(img, area.x, area.y, area.width, area.height, 0, 0, canvas.width, canvas.height);
   142→  const blob = await new Promise<Blob | null>((resolve) =>
   143→    canvas.toBlob(resolve, "image/jpeg", 0.92)
   144→  );
   145→  if (!blob) throw new Error("Couldn't process the cropped image");
   146→  return new File([blob], `cover-${Date.now()}.jpg`, { type: "image/jpeg" });
   147→}

   201→/** Maps a server error message back to the wizard step that most likely caused it. */
   202→function stepForMessage(msg: string): number {
   203→  const m = msg.toLowerCase();
   204→  if (/technolog/.test(m)) return 5;
   205→  if (/(member|teammate|username)/.test(m)) return 4;
   206→  if (/(cover|gallery|image)/.test(m)) return 2;
   207→  if (/(department|category|semester|supervisor|program|academic|year)/.test(m)) return 3;
   208→  if (/(url|resource|timeline|milestone|date|link)/.test(m)) return 6;
   209→  return 1; // title / summary / description and anything generic
   210→}

   341→/** Debounced user search + per-row role select, shared by the team step. */
   342→function UserSearchAdd({
   343→  exclude,
   344→  onPick,
   345→}: {
   346→  exclude: Set<string>;
   347→  onPick: (u: SafeUserDTO, role: string) => void;
   348→}) {
   349→  const [query, setQuery] = useState("");
   350→  const [debounced, setDebounced] = useState("");
   351→  const [found, setFound] = useState<SafeUserDTO[] | null>(null);
   352→  const [foundFor, setFoundFor] = useState("");
   353→  const [roles, setRoles] = useState<Record<string, string>>({});
   354→  const [busyName, setBusyName] = useState<string | null>(null);
   355→
   356→  useEffect(() => {
   357→    const t = setTimeout(() => setDebounced(query.trim().toLowerCase()), 300);
   358→    return () => clearTimeout(t);
   359→  }, [query]);
   360→
   361→  useEffect(() => {
   362→    const q = debounced;
   363→    if (q.length < 2) return;
   364→    const ac = new AbortController();
   365→    searchUsers(q, ac.signal)
   366→      .then((rows) => {
   367→        setFound(rows);
   368→        setFoundFor(q);
   369→      })
   370→      .catch(() => {});
   371→    return () => ac.abort();
   372→  }, [debounced]);
   373→
   374→  const ready = debounced.length >= 2 && foundFor === debounced;
   375→  const visible = ready ? (found ?? []).filter((u) => !exclude.has(u.username.toLowerCase())) : null;
   376→
   377→  const pick = (u: SafeUserDTO) => {
   378→    setBusyName(u.id);
   379→    setQuery("");
   380→    Promise.resolve(onPick(u, roles[u.id] ?? "MEMBER"))
   381→      .catch(() => {})
   382→      .finally(() => setBusyName(null));
   383→  };
   384→
   385→  return (
   386→    <div>
   387→      <Label htmlFor="team-search">Find teammates</Label>
   388→      <div className="relative mt-1.5">
   389→        <Search
   390→          className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   391→          aria-hidden
   392→        />
   393→        <Input
   394→          id="team-search"
   395→          value={query}
   396→          onChange={(e) => setQuery(e.target.value)}
   397→          placeholder="Search students by name or username…"
   398→          className="h-10 rounded-xl bg-white pl-9"
   399→          autoComplete="off"
   400→        />
   401→      </div>
   402→      <div className="mt-2 max-h-56 space-y-1.5 overflow-y-auto nice-scroll">
   403→        {visible === null ? (
   404→          <p className="px-1 py-2 text-xs text-muted-foreground">
   405→            {debounced.length >= 2 ? "Searching…" : "Type at least 2 characters to search."}
   406→          </p>
   407→        ) : visible.length === 0 ? (
   408→          <p className="px-1 py-2 text-xs text-muted-foreground">No matching students.</p>
   409→        ) : (
   410→          visible.map((u) => (
   411→            <div
   412→              key={u.id}
   413→              className="flex items-center gap-2.5 rounded-xl border border-up-border bg-white px-3 py-2"
   414→            >
   415→              <MemberAvatar name={u.displayName} ring={false} className="size-7" />
   416→              <div className="min-w-0 flex-1">
   417→                <p className="truncate text-sm font-medium text-foreground">{u.displayName}</p>
   418→                <p className="truncate text-xs text-muted-foreground">@{u.username}</p>
   419→              </div>
   420→              <Select value={roles[u.id] ?? "MEMBER"} onValueChange={(role) => setRoles((r) => ({ ...r, [u.id]: role }))}>
   421→                <SelectTrigger
   422→                  aria-label={`Role for ${u.displayName}`}
   423→                  className="h-8 w-[92px] rounded-lg text-xs"
   424→                >
   425→                  <SelectValue />
   426→                </SelectTrigger>
   427→                <SelectContent>
   428→                  <SelectItem value="LEAD">Lead</SelectItem>
   429→                  <SelectItem value="MEMBER">Member</SelectItem>
   430→                </SelectContent>
   431→              </Select>
   432→              <Button
   433→                size="sm"
   434→                className="h-8 rounded-lg px-3 font-semibold"
   435→                disabled={busyName === u.id}
   436→                onClick={() => pick(u)}
   437→              >
   438→                {busyName === u.id ? <Loader2 className="size-3.5 animate-spin" aria-hidden /> : "Add"}
   439→              </Button>
   440→            </div>
   441→          ))
   442→        )}
   443→      </div>
   444→    </div>
   445→  );
   446→}
