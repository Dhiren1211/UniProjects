   149→/** Uppercase file extension of an /uploads/… URL, e.g. "PDF". */
   150→function extOf(url: string) {
   151→  const m = url.match(/\.([a-z0-9]{1,5})(?:[?#]|$)/i);
   152→  return m ? m[1].toUpperCase() : "FILE";
   153→}
   154→
   155→/** Decoded basename of an /uploads/… path (falls back to the raw string). */
   156→function uploadBaseName(url: string) {
   157→  const last = url.split("/").pop() ?? url;
   158→  try {
   159→    return decodeURIComponent(last);
   160→  } catch {
   161→    return last;
   162→  }
   163→}

   165→function fmtDay(iso: string) {
   166→  try {

   168→  } catch {
   169→    return "";
   170→  }
   171→}

   543→function ManageTeamDialog({
   544→  open,
   545→  onOpenChange,
   546→  slug,
   547→  members,
   548→  viewerUsername,
   549→  onChanged,
   550→}: {
   551→  open: boolean;
   552→  onOpenChange: (v: boolean) => void;
   553→  slug: string;
   554→  members: MemberDTO[];
   555→  viewerUsername: string;
   556→  onChanged: () => void;
   557→}) {
   558→  const [query, setQuery] = useState("");
   559→  const [debounced, setDebounced] = useState("");
   560→  const [found, setFound] = useState<SafeUserDTO[] | null>(null);
   561→  const [foundFor, setFoundFor] = useState("");
   562→  const [roles, setRoles] = useState<Record<string, string>>({});
   563→  const [busyUser, setBusyUser] = useState<string | null>(null);
   564→  const [busyMember, setBusyMember] = useState<string | null>(null);
   565→  const [removing, setRemoving] = useState<MemberDTO | null>(null);
   566→
   567→  useEffect(() => {
   568→    const t = setTimeout(() => setDebounced(query.trim().toLowerCase()), 300);
   569→    return () => clearTimeout(t);
   570→  }, [query]);
   571→
   572→  useEffect(() => {
   573→    const q = debounced;
   574→    if (q.length < 2) return;
   575→    const ac = new AbortController();
   576→    searchUsers(q, ac.signal)
   577→      .then((rows) => {
   578→        setFound(rows);
   579→        setFoundFor(q);
   580→      })
   581→      .catch(() => {});
   582→    return () => ac.abort();
   583→  }, [debounced]);
   584→
   585→  const existing = new Set(
   586→    members.map((m) => m.username.toLowerCase()).concat(viewerUsername.toLowerCase())
   587→  );
   588→  const ready = debounced.length >= 2 && foundFor === debounced;
   589→  const visible = ready
   590→    ? (found ?? []).filter((u) => !existing.has(u.username.toLowerCase()))
   591→    : null;
   592→
   593→  const changeRole = (m: MemberDTO, role: string) => {
   594→    if (role === m.role) return;
   595→    setBusyMember(m.id);
   596→    updateMember(slug, { memberId: m.id, role })
   597→      .then(() => {
   598→        toast.success(`${m.name} is now ${ROLE_LABEL[role] ?? role.toLowerCase()}`);
