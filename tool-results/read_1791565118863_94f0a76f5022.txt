   212→/** "/project/<slug>" / "/community" → internal view, anything else is not navigable. */
   213→function activityToView(href: string): View | null {
   214→  const project = /^\/project\/([^/?#]+)/.exec(href);
   215→  if (project) return { name: "project", slug: project[1] };
   216→  if (href.startsWith("/community")) return { name: "community" };
   217→  return null;
   218→}

   220→/** "2025-11" → "Nov". */
   221→function monthLabel(key: string) {
   222→  const [year, month] = key.split("-");
   223→  const idx = Number(month) - 1;
   224→  if (!year || !Number.isFinite(idx) || idx < 0 || idx > 11) return key;
   225→  return format(new Date(Number(year), idx, 1), "MMM");
   226→}

   228→/** value → debounced value (300 ms) — keeps search/filters off the network until typing settles. */
   229→function useDebounced<T>(value: T, delay = 300): T {
   230→  const [debounced, setDebounced] = useState(value);
   231→  useEffect(() => {
   232→    const t = setTimeout(() => setDebounced(value), delay);
   233→    return () => clearTimeout(t);
   234→  }, [value, delay]);
   235→  return debounced;
   236→}

   242→export function AdminView({ navigate }: { navigate: Navigate }) {
   243→  const { user, authLoaded, openAuth } = useAppStore();
   244→  const reduceMotion = useReducedMotion();
   245→  const [tab, setTab] = useState<TabKey>("overview");
   246→  const [overview, setOverview] = useState<AdminOverviewDTO | null>(null);
   247→  const [overviewError, setOverviewError] = useState(false);
   248→  const [refreshTick, setRefreshTick] = useState(0);
   249→  const [refreshing, setRefreshing] = useState(false);
   250→  const [loadedAt, setLoadedAt] = useState<Partial<Record<TabKey, number>>>({});
   251→  const [arriveNew, setArriveNew] = useState(false);
   252→
   253→  const isAdmin = user?.role === "ADMIN";
   254→
   255→  // Keep the "Updated …" captions honest without refetching anything.
   256→  const [, setClock] = useState(0);
   257→  useEffect(() => {
   258→    const t = setInterval(() => setClock((n) => n + 1), 30_000);
   259→    return () => clearInterval(t);
   260→  }, []);
   261→
   262→  /** Record a finished load: always stops the spinner; timestamps only on success. */
   263→  const noteLoaded = useCallback((keys: TabKey[], ok: boolean) => {
   264→    setRefreshing(false);
   265→    if (!ok) return;
   266→    const at = Date.now();
   267→    setLoadedAt((prev) => {
   268→      const next = { ...prev };
   269→      for (const key of keys) next[key] = at;
   270→      return next;
   271→    });
   272→  }, []);
   273→
   274→  // Stable per-tab onLoaded callbacks — tabs include them in their fetch-effect
   275→  // deps, so a fresh identity per render would trigger a refetch loop.
   276→  const overviewLoaded = useCallback(
   277→    (ok: boolean) => noteLoaded(["overview", "moderation"], ok),
   278→    [noteLoaded]
   279→  );
   280→  const projectsLoaded = useCallback((ok: boolean) => noteLoaded(["projects"], ok), [noteLoaded]);
   281→  const usersLoaded = useCallback((ok: boolean) => noteLoaded(["users"], ok), [noteLoaded]);
   282→  const departmentsLoaded = useCallback(
   283→    (ok: boolean) => noteLoaded(["departments"], ok),
   284→    [noteLoaded]
   285→  );
   286→  const eventsLoaded = useCallback((ok: boolean) => noteLoaded(["events"], ok), [noteLoaded]);
   287→
   288→  /** Explicit refresh from the content header — spins the button until data lands. */
   289→  const refresh = () => {
   290→    setRefreshing(true);
   291→    setRefreshTick((n) => n + 1);
   292→  };
   293→
   294→  /** Background re-sync of shared overview data after a mutation (no spinner). */
   295→  const syncOverview = () => setRefreshTick((n) => n + 1);
   296→
   297→  /** Switch tabs; quick actions can request the tab's "New …" dialog to open on arrival. */
   298→  const goTab = (key: TabKey, openNew = false) => {
   299→    setTab(key);
   300→    setArriveNew(openNew);
   301→  };
   302→
   303→  useEffect(() => {
   304→    if (!authLoaded || !isAdmin) return;
   305→    let cancelled = false;
   306→    fetchAdminOverview()
   307→      .then((data) => {
   308→        if (cancelled) return;
   309→        setOverview(data);
   310→        setOverviewError(false);
   311→        overviewLoaded(true);
   312→      })
   313→      .catch((e) => {
   314→        if (cancelled) return;
   315→        setOverviewError(true);
   316→        overviewLoaded(false);
   317→        toast.error(e instanceof Error ? e.message : "Could not load the admin overview");
   318→      });
   319→    return () => {
   320→      cancelled = true;
   321→    };
   322→  }, [refreshTick, authLoaded, isAdmin, overviewLoaded]);
   323→
   324→  // ---------- gates ----------
   325→
   326→  if (!authLoaded) {
   327→    return (
   328→      <Shell>
   329→        <div className="space-y-6">
   330→          <Skeleton className="h-24 w-full rounded-2xl" />
   331→          <Skeleton className="h-14 w-full rounded-2xl" />
   332→          <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   333→            {Array.from({ length: 8 }).map((_, i) => (
   334→              <Skeleton key={i} className="h-28 w-full rounded-2xl" />
   335→            ))}
   336→          </div>
   337→        </div>
   338→      </Shell>
   339→    );
   340→  }
   341→
   342→  if (!user || user.role !== "ADMIN") {
   343→    return (
   344→      <Shell>
   345→        <EmptyState
   346→          icon={user ? ShieldAlert : Lock}
   347→          title="Admin access required"
   348→          hint="This console manages the whole showcase — projects, members, departments and events. It is limited to Kyungsung University administrators."
   349→          action={
   350→            !user ? (
   351→              <Button onClick={() => openAuth("login")} className="rounded-xl font-semibold">
   352→                Sign in
   353→              </Button>
   354→            ) : undefined
   355→          }
   356→        />
   357→      </Shell>
   358→    );
   359→  }
   360→
   361→  const moderationPending = (overview?.pendingVerifications ?? 0) + (overview?.openReports ?? 0);
   362→  const meta = TAB_META[tab];
   363→
   364→  return (
   365→    <div className="mx-auto w-full max-w-7xl px-5 py-6 pb-20 md:px-8 md:py-8">
   366→      <div className="lg:grid lg:grid-cols-[15rem_minmax(0,1fr)] lg:gap-6 xl:gap-8">
   367→        <ConsoleSidebar
   368→          tab={tab}
   369→          overview={overview}
   370→          moderationPending={moderationPending}
   371→          user={user}
   372→          onTab={goTab}
   373→          navigate={navigate}
   374→        />
   375→
   376→        <main className="min-w-0">
   377→          {/* < lg: brand + scrollable pills replace the sidebar */}
   378→          <ConsoleTopBar
   379→            tab={tab}
   380→            moderationPending={moderationPending}
   381→            user={user}
   382→            onTab={goTab}
   383→          />
   384→
   385→          {/* < lg: the sidebar system mini-card lives at the top of Overview */}
   386→          {tab === "overview" && overview ? (
   387→            <SystemMiniCard storage={overview.storage} className="mt-4 lg:hidden" />
   388→          ) : null}
   389→
   390→          <ContentHeader
   391→            title={meta.title}
   392→            description={meta.description}
   393→            loadedAt={loadedAt[tab]}
   394→            refreshing={refreshing}
   395→            onRefresh={refresh}
   396→            className="mt-6"
   397→          />
   398→
   399→          {/* ---------- active tab ---------- */}
   400→          <div className="mt-5">
   401→            <AnimatePresence mode="wait" initial={false}>
   402→              <motion.div
   403→                key={tab}
   404→                initial={reduceMotion ? false : { opacity: 0, y: 10 }}
   405→                animate={{ opacity: 1, y: 0 }}
   406→                exit={reduceMotion ? { opacity: 0 } : { opacity: 0, y: -8 }}
   407→                transition={{ duration: 0.18, ease: "easeOut" }}
   408→              >
   409→            {tab === "overview" ? (
   410→              <OverviewTab
   411→                overview={overview}
   412→                error={overviewError}
   413→                onRetry={refresh}
   414→                navigate={navigate}
   415→                onQuickAction={goTab}
   416→              />
   417→            ) : null}
   418→            {tab === "projects" ? (
   419→              <ProjectsTab
   420→                navigate={navigate}
   421→                onChanged={syncOverview}
   422→                refreshTick={refreshTick}
   423→                onLoaded={projectsLoaded}
   424→              />
   425→            ) : null}
   426→            {tab === "users" ? (
   427→              <UsersTab
   428→                onChanged={syncOverview}
   429→                refreshTick={refreshTick}
   430→                onLoaded={usersLoaded}
   431→                autoOpenNew={arriveNew}
   432→              />
   433→            ) : null}
   434→            {tab === "departments" ? (
   435→              <DepartmentsTab
   436→                onChanged={syncOverview}
   437→                refreshTick={refreshTick}
   438→                onLoaded={departmentsLoaded}
   439→                autoOpenNew={arriveNew}
   440→              />
   441→            ) : null}
   442→            {tab === "events" ? (
   443→              <EventsTab
   444→                onChanged={syncOverview}
   445→                refreshTick={refreshTick}
   446→                onLoaded={eventsLoaded}
   447→                autoOpenNew={arriveNew}
   448→              />
   449→            ) : null}
   450→            {tab === "moderation" ? <ModerationTab overview={overview} navigate={navigate} /> : null}
   451→              </motion.div>
   452→            </AnimatePresence>
   453→          </div>
   454→        </main>
   455→      </div>
   456→    </div>
   457→  );
   458→}

   800→/** rAF count-up that respects reduced motion — gives every number a live "landing" feel. */
   801→function useCountUp(target: number, duration = 900): number {
   802→  const reduceMotion = useReducedMotion();
   803→  const [value, setValue] = useState(0);
   804→  useEffect(() => {
   805→    if (reduceMotion) return; // derived below — no animation needed
   806→    let raf = 0;
   807→    const start = performance.now();
   808→    const tick = (now: number) => {
   809→      const t = Math.min(1, (now - start) / duration);
   810→      const eased = 1 - Math.pow(1 - t, 3); // ease-out cubic
   811→      setValue(Math.round(target * eased));
   812→      if (t < 1) raf = requestAnimationFrame(tick);
   813→    };
   814→    raf = requestAnimationFrame(tick);
   815→    return () => cancelAnimationFrame(raf);
   816→  }, [target, duration, reduceMotion]);
   817→  return reduceMotion ? target : value;
   818→}
