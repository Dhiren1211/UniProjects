   219→  3050→/** Memorable 12-char password guaranteed to mix upper, lower and digits. */
   220→  3051→function generatePassword(): string {
   221→  3052→  const lower = "abcdefghijkmnpqrstuvwxyz";
   222→  3053→  const upper = "ABCDEFGHJKLMNPQRSTUVWXYZ";
   223→  3054→  const digits = "23456789";
   224→  3055→  const all = lower + upper + digits;
   225→  3056→  const pick = (set: string) => set[Math.floor(Math.random() * set.length)];
   226→  3057→  const out = [pick(upper), pick(lower), pick(digits)];
   227→  3058→  for (let i = 0; i < 9; i++) out.push(pick(all));
   228→  3059→  for (let i = out.length - 1; i > 0; i--) {
   229→  3060→    const j = Math.floor(Math.random() * (i + 1));
   230→  3061→    [out[i], out[j]] = [out[j], out[i]];
   231→  3062→  }
   232→  3063→  return out.join("");
   233→  3064→}
