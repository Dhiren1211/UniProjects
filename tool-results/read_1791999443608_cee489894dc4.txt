   201→/** "/project/<slug>" / "/community" → internal view, anything else is not navigable. */
   202→function activityToView(href: string): View | null {
   203→  const project = /^\/project\/([^/?#]+)/.exec(href);
   204→  if (project) return { name: "project", slug: project[1] };
   205→  if (href.startsWith("/community")) return { name: "community" };
   206→  return null;
   207→}

   209→/** "2025-11" → "Nov". */
   210→function monthLabel(key: string) {
   211→  const [year, month] = key.split("-");
   212→  const idx = Number(month) - 1;
   213→  if (!year || !Number.isFinite(idx) || idx < 0 || idx > 11) return key;
   214→  return format(new Date(Number(year), idx, 1), "MMM");
   215→}

   217→/** value → debounced value (300 ms) — keeps search/filters off the network until typing settles. */
   218→function useDebounced<T>(value: T, delay = 300): T {
   219→  const [debounced, setDebounced] = useState(value);
   220→  useEffect(() => {
   221→    const t = setTimeout(() => setDebounced(value), delay);
   222→    return () => clearTimeout(t);
   223→  }, [value, delay]);
   224→  return debounced;
   225→}

   231→export function AdminView({ navigate }: { navigate: Navigate }) {
   232→  const { user, authLoaded, openAuth } = useAppStore();
   233→  const reduceMotion = useReducedMotion();
   234→  const [tab, setTab] = useState<TabKey>("overview");
   235→  const [overview, setOverview] = useState<AdminOverviewDTO | null>(null);
   236→  const [overviewError, setOverviewError] = useState(false);
   237→  const [refreshTick, setRefreshTick] = useState(0);
   238→  const [refreshing, setRefreshing] = useState(false);
   239→  const [loadedAt, setLoadedAt] = useState<Partial<Record<TabKey, number>>>({});
   240→  const [arriveNew, setArriveNew] = useState(false);
   241→
   242→  const isAdmin = user?.role === "ADMIN";
   243→
   244→  // Keep the "Updated …" captions honest without refetching anything.
   245→  const [, setClock] = useState(0);
   246→  useEffect(() => {
   247→    const t = setInterval(() => setClock((n) => n + 1), 30_000);
   248→    return () => clearInterval(t);
   249→  }, []);
   250→
   251→  /** Record a finished load: always stops the spinner; timestamps only on success. */
   252→  const noteLoaded = useCallback((keys: TabKey[], ok: boolean) => {
   253→    setRefreshing(false);
   254→    if (!ok) return;
   255→    const at = Date.now();
   256→    setLoadedAt((prev) => {
   257→      const next = { ...prev };
   258→      for (const key of keys) next[key] = at;
   259→      return next;
   260→    });
   261→  }, []);
   262→
   263→  // Stable per-tab onLoaded callbacks — tabs include them in their fetch-effect
   264→  // deps, so a fresh identity per render would trigger a refetch loop.
   265→  const overviewLoaded = useCallback(
   266→    (ok: boolean) => noteLoaded(["overview", "moderation"], ok),
   267→    [noteLoaded]
   268→  );
   269→  const projectsLoaded = useCallback((ok: boolean) => noteLoaded(["projects"], ok), [noteLoaded]);
   270→  const usersLoaded = useCallback((ok: boolean) => noteLoaded(["users"], ok), [noteLoaded]);
   271→  const departmentsLoaded = useCallback(
   272→    (ok: boolean) => noteLoaded(["departments"], ok),
   273→    [noteLoaded]
   274→  );
   275→  const eventsLoaded = useCallback((ok: boolean) => noteLoaded(["events"], ok), [noteLoaded]);
   276→
   277→  /** Explicit refresh from the content header — spins the button until data lands. */
   278→  const refresh = () => {
   279→    setRefreshing(true);
   280→    setRefreshTick((n) => n + 1);
   281→  };
   282→
   283→  /** Background re-sync of shared overview data after a mutation (no spinner). */
   284→  const syncOverview = () => setRefreshTick((n) => n + 1);
   285→
   286→  /** Switch tabs; quick actions can request the tab's "New …" dialog to open on arrival. */
   287→  const goTab = (key: TabKey, openNew = false) => {
   288→    setTab(key);
   289→    setArriveNew(openNew);
   290→    window.scrollTo({ top: 0 });
   291→  };
   292→
   293→  useEffect(() => {
   294→    if (!authLoaded || !isAdmin) return;
   295→    let cancelled = false;
   296→    fetchAdminOverview()
   297→      .then((data) => {
   298→        if (cancelled) return;
   299→        setOverview(data);
   300→        setOverviewError(false);
   301→        overviewLoaded(true);
   302→      })
   303→      .catch((e) => {
   304→        if (cancelled) return;
   305→        setOverviewError(true);
   306→        overviewLoaded(false);
   307→        toast.error(e instanceof Error ? e.message : "Could not load the admin overview");
   308→      });
   309→    return () => {
   310→      cancelled = true;
   311→    };
   312→  }, [refreshTick, authLoaded, isAdmin, overviewLoaded]);
   313→
   314→  // ---------- gates ----------
   315→
   316→  if (!authLoaded) {
   317→    return (
   318→      <Shell>
   319→        <div className="space-y-6">
   320→          <Skeleton className="h-24 w-full rounded-2xl" />
   321→          <Skeleton className="h-14 w-full rounded-2xl" />
   322→          <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   323→            {Array.from({ length: 8 }).map((_, i) => (
   324→              <Skeleton key={i} className="h-28 w-full rounded-2xl" />
   325→            ))}
   326→          </div>
   327→        </div>
   328→      </Shell>
   329→    );
   330→  }
   331→
   332→  if (!user || user.role !== "ADMIN") {
   333→    return (
   334→      <Shell>
   335→        <EmptyState
   336→          icon={user ? ShieldAlert : Lock}
   337→          title="Admin access required"
   338→          hint="This console manages the whole showcase — projects, members, departments and events. It is limited to Kyungsung University administrators."
   339→          action={
   340→            !user ? (
   341→              <Button onClick={() => openAuth("login")} className="rounded-xl font-semibold">
   342→                Sign in
   343→              </Button>
   344→            ) : undefined
   345→          }
   346→        />
   347→      </Shell>
   348→    );
   349→  }
   350→
   351→  const moderationPending = (overview?.pendingVerifications ?? 0) + (overview?.openReports ?? 0);
   352→  const meta = TAB_META[tab];
   353→
   354→  return (
   355→    <div className="lg:flex">
   356→      <ConsoleSidebar
   357→        tab={tab}
   358→        moderationPending={moderationPending}
   359→        onTab={goTab}
   360→        navigate={navigate}
   361→      />
   362→
   363→      <main className="min-w-0 flex-1 px-4 py-5 md:px-6 md:py-6">
   364→          {/* < lg: brand + scrollable pills replace the sidebar */}
   365→          <ConsoleTopBar
   366→            tab={tab}
   367→            moderationPending={moderationPending}
   368→            user={user}
   369→            onTab={goTab}
   370→          />
   371→
   372→          {/* < lg: the sidebar system mini-card lives at the top of Overview */}
   373→          {tab === "overview" && overview ? (
   374→            <SystemMiniCard storage={overview.storage} className="mt-4 lg:hidden" />
   375→          ) : null}
   376→
   377→          <ContentHeader
   378→            title={meta.title}
   379→            description={meta.description}
   380→            loadedAt={loadedAt[tab]}
   381→            refreshing={refreshing}
   382→            onRefresh={refresh}
   383→            className="mt-5"
   384→          />
   385→
   386→          {/* ---------- active tab ---------- */}
   387→          <div className="mt-4">
   388→            <AnimatePresence mode="wait" initial={false}>
   389→              <motion.div
   390→                key={tab}
   391→                initial={reduceMotion ? false : { opacity: 0, y: 10 }}
   392→                animate={{ opacity: 1, y: 0 }}
   393→                exit={reduceMotion ? { opacity: 0 } : { opacity: 0, y: -8 }}
   394→                transition={{ duration: 0.18, ease: "easeOut" }}
   395→              >
   396→            {tab === "overview" ? (
   397→              <OverviewTab
   398→                overview={overview}
   399→                error={overviewError}
   400→                onRetry={refresh}
   401→                navigate={navigate}
   402→                onQuickAction={goTab}
   403→              />
   404→            ) : null}
   405→            {tab === "projects" ? (
   406→              <ProjectsTab
   407→                navigate={navigate}
   408→                onChanged={syncOverview}
   409→                refreshTick={refreshTick}
   410→                onLoaded={projectsLoaded}
   411→              />
   412→            ) : null}
   413→            {tab === "users" ? (
   414→              <UsersTab
   415→                onChanged={syncOverview}
   416→                refreshTick={refreshTick}
   417→                onLoaded={usersLoaded}
   418→                autoOpenNew={arriveNew}
   419→              />
   420→            ) : null}
   421→            {tab === "departments" ? (
   422→              <DepartmentsTab
   423→                onChanged={syncOverview}
   424→                refreshTick={refreshTick}
   425→                onLoaded={departmentsLoaded}
   426→                autoOpenNew={arriveNew}
   427→              />
   428→            ) : null}
   429→            {tab === "events" ? (
   430→              <EventsTab
   431→                onChanged={syncOverview}
   432→                refreshTick={refreshTick}
   433→                onLoaded={eventsLoaded}
   434→                autoOpenNew={arriveNew}
   435→              />
   436→            ) : null}
   437→            {tab === "moderation" ? <ModerationTab overview={overview} navigate={navigate} /> : null}
   438→              </motion.div>
   439→            </AnimatePresence>
   440→          </div>
   441→      </main>
   442→    </div>
   443→  );
   444→}

   541→      </div>
   542→    </aside>
   543→  );

   690→/**
   691→ * Fixed-size page window — the pager always renders the same slots:
   692→ *   edges:  `1 2 3 4 5 … N`  /  `1 … N-4 N-3 N-2 N-1 N`
   693→ *   middle: `1 … p-1 p p+1 … N`
   694→ * so its width never changes while navigating (e.g. `1 2 3 … 100`).
   695→ */
   696→function pageList(page: number, totalPages: number): (number | "…")[] {
   697→  if (totalPages <= 5) return Array.from({ length: totalPages }, (_, i) => i + 1);
   698→  if (page <= 3) return [1, 2, 3, 4, 5, "…", totalPages];
   699→  if (page >= totalPages - 2)
   700→    return [1, "…", totalPages - 4, totalPages - 3, totalPages - 2, totalPages - 1, totalPages];
   701→  return [1, "…", page - 1, page, page + 1, "…", totalPages];
   702→}

   892→/** rAF count-up that respects reduced motion — gives every number a live "landing" feel. */
   893→function useCountUp(target: number, duration = 900): number {
   894→  const reduceMotion = useReducedMotion();
   895→  const [value, setValue] = useState(0);
   896→  useEffect(() => {
   897→    if (reduceMotion) return; // derived below — no animation needed
   898→    let raf = 0;
   899→    const start = performance.now();
   900→    const tick = (now: number) => {
   901→      const t = Math.min(1, (now - start) / duration);
   902→      const eased = 1 - Math.pow(1 - t, 3); // ease-out cubic
   903→      setValue(Math.round(target * eased));
   904→      if (t < 1) raf = requestAnimationFrame(tick);
   905→    };
   906→    raf = requestAnimationFrame(tick);
   907→    return () => cancelAnimationFrame(raf);
   908→  }, [target, duration, reduceMotion]);
   909→  return reduceMotion ? target : value;
   910→}

  1404→/** "Member growth" — hand-rolled SVG area chart of monthly signups (reference dashboard style). */
  1405→function GrowthChartCard({
  1406→  data,
  1407→  className,
  1408→}: {
  1409→  data: { month: string; count: number }[];
  1410→  className?: string;
  1411→}) {
  1412→  const reduceMotion = useReducedMotion();
  1413→  const W = 560;
  1414→  const H = 230;
  1415→  const PL = 36;
  1416→  const PR = 14;
  1417→  const PT = 16;
  1418→  const PB = 30;
  1419→  const iw = W - PL - PR;
  1420→  const ih = H - PT - PB;
  1421→  const max = Math.max(1, ...data.map((d) => d.count));
  1422→  const top = Math.max(4, Math.ceil(max / 4) * 4);
  1423→  const total = data.reduce((sum, d) => sum + d.count, 0);
  1424→  const x = (i: number) => PL + (data.length <= 1 ? iw / 2 : (i / (data.length - 1)) * iw);
  1425→  const y = (v: number) => PT + ih - (v / top) * ih;
  1426→  const linePath = data.length
  1427→    ? `M${data.map((d, i) => `${x(i).toFixed(1)},${y(d.count).toFixed(1)}`).join(" L")}`
  1428→    : "";
  1429→  const areaPath = data.length
  1430→    ? `${linePath} L${x(data.length - 1).toFixed(1)},${(PT + ih).toFixed(1)} L${x(0).toFixed(1)},${(PT + ih).toFixed(1)} Z`
  1431→    : "";
  1432→  const ticks = [0, 1, 2, 3, 4].map((i) => (top / 4) * i);
  1433→
  1434→  return (
  1435→    <Panel
  1436→      title="Member growth"
  1437→      caption={`New signups per month · ${total} total`}
  1438→      className={className}
  1439→      right={
  1440→        <span className="inline-flex shrink-0 items-center gap-1.5 rounded-full bg-primary-50 px-2.5 py-1 text-[11px] font-bold text-primary-700">

  1443→        </span>
  1444→      }

  1447→        <p className="mt-5 text-sm text-muted-foreground">No signup data yet.</p>
  1448→      ) : (
  1449→        <svg
  1450→          viewBox={`0 0 ${W} ${H}`}
  1451→          className="mt-2 h-auto w-full"
  1452→          role="img"
  1453→          aria-label={`Area chart of monthly signups, ${total} total`}
  1454→        >
  1455→          <defs>
  1456→            <linearGradient id="growth-fill" x1="0" y1="0" x2="0" y2="1">
  1457→              <stop offset="0%" stopColor="var(--up-primary-500)" stopOpacity="0.28" />
  1458→              <stop offset="100%" stopColor="var(--up-primary-500)" stopOpacity="0.02" />
  1459→            </linearGradient>
  1460→          </defs>
  1461→          {ticks.map((t) => (
  1462→            <g key={t}>
  1463→              <line
  1464→                x1={PL}
  1465→                x2={W - PR}
  1466→                y1={y(t)}
  1467→                y2={y(t)}
  1468→                stroke="var(--up-border)"
  1469→                strokeDasharray="3 4"
  1470→              />
  1471→              <text x={PL - 8} y={y(t) + 3.5} textAnchor="end" fontSize="10" fill="var(--up-ink-500)">
  1472→                {t}
  1473→              </text>
  1474→            </g>
  1475→          ))}
  1476→          {reduceMotion ? (
  1477→            <>
  1478→              <path d={areaPath} fill="url(#growth-fill)" />
  1479→              <path
  1480→                d={linePath}
  1481→                fill="none"
  1482→                stroke="var(--up-primary-500)"
  1483→                strokeWidth="2.5"
  1484→                strokeLinecap="round"
  1485→                strokeLinejoin="round"
  1486→              />
  1487→            </>
  1488→          ) : (
  1489→            <>
  1490→              <motion.path
  1491→                d={areaPath}
  1492→                fill="url(#growth-fill)"
  1493→                initial={{ opacity: 0 }}
  1494→                animate={{ opacity: 1 }}
  1495→                transition={{ duration: 0.7, delay: 0.3 }}
  1496→              />
  1497→              <motion.path
  1498→                d={linePath}
  1499→                fill="none"
  1500→                stroke="var(--up-primary-500)"
  1501→                strokeWidth="2.5"
  1502→                strokeLinecap="round"
  1503→                strokeLinejoin="round"
  1504→                initial={{ pathLength: 0 }}
  1505→                animate={{ pathLength: 1 }}
  1506→                transition={{ duration: 0.9, ease: "easeOut" }}
  1507→              />
  1508→            </>
  1509→          )}
  1510→          {data.map((d, i) => (
  1511→            <g key={d.month}>
  1512→              <circle
  1513→                cx={x(i)}
  1514→                cy={y(d.count)}
  1515→                r="3.5"
  1516→                fill="#fff"
  1517→                stroke="var(--up-primary-500)"
  1518→                strokeWidth="2"

  1540→/** "Community make-up" — SVG donut of members by role with center total + legend. */
  1541→function RoleDonutCard({
  1542→  overview,
  1543→  className,
  1544→}: {
  1545→  overview: AdminOverviewDTO;
  1546→  className?: string;
  1547→}) {
  1548→  const admins = Math.max(0, overview.users - overview.students - overview.faculty);
  1549→  const slices = [
  1550→    { label: "Students", value: overview.students, color: "var(--up-primary-500)" },
  1551→    { label: "Faculty & staff", value: overview.faculty, color: "var(--up-info)" },
  1552→    { label: "Admins", value: admins, color: "#fcaf17" },
  1553→  ].filter((s) => s.value > 0);
  1554→  const total = slices.reduce((sum, s) => sum + s.value, 0);
  1555→
  1556→  // dasharray on a 100-unit circumference (r = 15.915) keeps the math readable
  1557→  const R = 15.915;

  1559→  const segs = slices.map((s) => {
  1560→    const len = (s.value / total) * 100;

  1563→    return seg;
  1564→  });

  1566→  return (
  1567→    <Panel title="Community make-up" caption="Members by role" className={className}>

  1576→                <circle
  1577→                  key={s.label}
  1578→                  cx="21"
  1579→                  cy="21"
  1580→                  r={R}
  1581→                  fill="none"
  1582→                  stroke={s.color}
  1583→                  strokeWidth="5.5"
  1584→                  strokeDasharray={`${s.len} ${100 - s.len}`}
  1585→                  strokeDashoffset={s.offset}
  1586→                >
  1587→                  <title>{`${s.label}: ${s.value}`}</title>

  1589→              ))}
  1590→            </svg>
  1591→            <div className="pointer-events-none absolute inset-0 flex flex-col items-center justify-center">

  1593→                {total.toLocaleString()}
