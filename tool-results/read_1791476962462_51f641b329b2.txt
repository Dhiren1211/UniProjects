    78→function CountUp({ value, className }: { value: number; className?: string }) {
    79→  const ref = useRef<HTMLSpanElement>(null);
    80→  const inView = useInView(ref, { once: true, margin: "-40px" });
    81→
    82→  useEffect(() => {
    83→    if (!inView || !ref.current) return;
    84→    const controls = animate(0, value, {
    85→      duration: 1.4,
    86→      ease: "easeOut",
    87→      onUpdate: (v) => {
    88→        if (ref.current) ref.current.textContent = Math.round(v).toLocaleString();
    89→      },
    90→    });
    91→    return () => controls.stop();
    92→  }, [inView, value]);
    93→
    94→  return (
    95→    <span ref={ref} className={className}>
    96→      0
    97→    </span>
    98→  );
    99→}

   101→function RotatingWord() {
   102→  const [idx, setIdx] = useState(0);
   103→
   104→  useEffect(() => {
   105→    const t = setInterval(() => setIdx((i) => (i + 1) % HEADLINE_WORDS.length), 2600);
   106→    return () => clearInterval(t);
   107→  }, []);
   108→
   109→  return (
   110→    <span className="relative inline-grid align-bottom">
   111→      <span className="sr-only">{HEADLINE_WORDS[0]}</span>
   112→      {HEADLINE_WORDS.map((w, i) => (
   113→        <motion.span
   114→          key={w}
   115→          aria-hidden
   116→          className="col-start-1 row-start-1 bg-gradient-to-r from-teal-300 via-teal-200 to-lime-200 bg-clip-text text-transparent"
   117→          initial={false}
   118→          animate={
   119→            i === idx
   120→              ? { opacity: 1, y: 0, filter: "blur(0px)" }
   121→              : { opacity: 0, y: 10, filter: "blur(4px)" }
   122→          }
   123→          transition={{ duration: 0.45, ease: "easeOut" }}
   124→        >
   125→          {w}
   126→        </motion.span>
   127→      ))}
   128→    </span>

   155→function HeroSearch({ navigate }: { navigate: Navigate }) {
   156→  const [q, setQ] = useState("");
   157→  const [results, setResults] = useState<SearchResults | null>(null);
   158→  const [open, setOpen] = useState(false);
   159→  const [loading, setLoading] = useState(false);
   160→  const boxRef = useRef<HTMLDivElement>(null);
   161→  const seqRef = useRef(0);
   162→
   163→  useEffect(() => {
   164→    const query = q.trim();
   165→    const t = setTimeout(() => {
   166→      if (query.length < 2) {
   167→        setResults(null);
   168→        setOpen(false);
   169→        setLoading(false);
   170→        return;
   171→      }
   172→      setLoading(true);
   173→      const seq = ++seqRef.current;
   174→      globalSearch(query)
   175→        .then((r) => {
   176→          if (seq !== seqRef.current) return;
   177→          setResults(r);
   178→          setOpen(true);
   179→        })
   180→        .catch(() => {})
   181→        .finally(() => {
   182→          if (seq === seqRef.current) setLoading(false);
   183→        });
   184→    }, 250);
   185→    return () => clearTimeout(t);
   186→  }, [q]);
   187→
   188→  useEffect(() => {
   189→    if (!open) return;
   190→    const onDocDown = (e: MouseEvent) => {
   191→      if (boxRef.current && !boxRef.current.contains(e.target as Node)) setOpen(false);
   192→    };
   193→    document.addEventListener("mousedown", onDocDown);
   194→    return () => document.removeEventListener("mousedown", onDocDown);
   195→  }, [open]);
   196→
   197→  const submit = () => {
   198→    setOpen(false);
   199→    navigate({ name: "projects", q: q.trim() || undefined });
   200→  };
   201→
   202→  const hasAny = results && (results.projects.length > 0 || results.users.length > 0);
   203→
   204→  return (
   205→    <div ref={boxRef} className="relative w-full max-w-xl">
   206→      {/* Bar */}
   207→      <div className="flex items-center gap-2 rounded-2xl border border-white/15 bg-white/10 p-2 shadow-float backdrop-blur-md transition-colors focus-within:border-teal-300/50">
   208→        <Search className="ml-2.5 size-4.5 shrink-0 text-teal-200/70" aria-hidden />
   209→        <input
   210→          value={q}
   211→          onChange={(e) => setQ(e.target.value)}
   212→          onFocus={() => {
   213→            if (results && q.trim().length >= 2) setOpen(true);
   214→          }}
   215→          onKeyDown={(e) => {
   216→            if (e.key === "Enter") submit();
   217→            if (e.key === "Escape") setOpen(false);
   218→          }}
   219→          placeholder="Search projects, people, tags…"
   220→          aria-label="Search the showcase"
   221→          className="h-9 min-w-0 flex-1 bg-transparent text-[15px] text-white outline-none placeholder:text-teal-100/50"
   222→        />
   223→        {loading ? <Loader2 className="size-4 shrink-0 animate-spin text-teal-200/70" aria-hidden /> : null}
   224→        <Button
   225→          onClick={submit}
   226→          className="h-9 rounded-xl bg-teal-400 px-4 text-sm font-semibold text-teal-950 shadow-none hover:bg-teal-300"
   227→        >
   228→          Search
   229→        </Button>
   230→      </div>
   231→
   232→      {/* Dropdown */}
   233→      {open ? (
   234→        <div className="absolute inset-x-0 top-full z-30 mt-2 overflow-hidden rounded-2xl border border-up-border bg-popover text-popover-foreground shadow-float">
   235→          {!hasAny ? (
   236→            <p className="px-5 py-6 text-center text-sm text-muted-foreground">
   237→              No matches for <span className="font-semibold text-foreground">“{q.trim()}”</span>
   238→            </p>
   239→          ) : (
   240→            <div className="max-h-80 overflow-y-auto nice-scroll">
   241→              {results!.projects.length > 0 ? (
   242→                <div className="p-2">
   243→                  <p className="px-3 pb-1.5 pt-2 text-[11px] font-semibold uppercase tracking-widest text-muted-foreground">
   244→                    Projects
   245→                  </p>
   246→                  {results!.projects.slice(0, 4).map((p) => (
   247→                    <button
   248→                      key={p.id}
   249→                      type="button"
   250→                      onClick={() => {
   251→                        setOpen(false);
   252→                        navigate({ name: "project", slug: p.slug });
   253→                      }}
   254→                      className="flex w-full items-center gap-3 rounded-xl px-3 py-2 text-left transition-colors hover:bg-surface-muted"
   255→                    >
   256→                      <span
   257→                        className={cn(
   258→                          "flex size-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br",
   259→                          accentOf(p.department.accent).cover

   263→                      </span>

   265→                        <span className="block truncate text-sm font-medium text-foreground">{p.title}</span>

   269→                        {p.department.code}
   270→                      </span>
   271→                    </button>
   272→                  ))}
   273→                </div>
   274→              ) : null}
   275→
   276→              {results!.users.length > 0 ? (
   277→                <div className="border-t border-up-border p-2">
   278→                  <p className="px-3 pb-1.5 pt-2 text-[11px] font-semibold uppercase tracking-widest text-muted-foreground">

   365→function FloatingCollage({
   366→  projects,
   367→  sx,
   368→  sy,
   369→  navigate,
   370→}: {
   371→  projects: ProjectListItem[] | null;
   372→  sx: ReturnType<typeof useSpring>;
   373→  sy: ReturnType<typeof useSpring>;
   374→  navigate: Navigate;

   376→  // Parallax depths (hooks must be unconditional)
   377→  const x1 = useTransform(sx, (v) => v * -30);

   380→  const y2 = useTransform(sy, (v) => v * 18);
   381→  const x3 = useTransform(sx, (v) => v * -14);
   382→  const y3 = useTransform(sy, (v) => v * 26);
   383→  const xChip = useTransform(sx, (v) => v * 40);
   384→  const yChip = useTransform(sy, (v) => v * -34);
   385→
   386→  const items = projects?.slice(0, 3) ?? [];
   387→  const students = items.flatMap((p) => p.members.slice(0, 2));
   388→
   389→  return (
   390→    <div className="relative mx-auto hidden h-[430px] w-full max-w-[540px] lg:block">
   391→      {/* rings */}
   392→      <div aria-hidden className="absolute left-1/2 top-1/2 size-[25rem] -translate-x-1/2 -translate-y-1/2 rounded-full border border-teal-200/10" />
   393→      <div aria-hidden className="absolute left-1/2 top-1/2 size-[33rem] -translate-x-1/2 -translate-y-1/2 rounded-full border border-teal-200/[0.06]" />

   400→                "absolute h-44 w-60 animate-pulse rounded-2xl bg-white/[0.06]",

   424→        <motion.div style={{ x: x3, y: y3 }} className="absolute bottom-10 left-[calc(50%-7.5rem)]">
   425→          <div className="[animation:up-float_9s_ease-in-out_-5s_infinite] rotate-1">
   426→            <CollageCard project={items[2]} onOpen={(slug) => navigate({ name: "project", slug })} />
   427→          </div>
   428→        </motion.div>

   431→      {/* floating chips */}
   432→      <motion.div style={{ x: xChip, y: yChip }} className="absolute right-4 top-2">
   433→        <div className="[animation:up-float_6s_ease-in-out_-1s_infinite]">

  1059→export function HomeView({ navigate }: { navigate: Navigate }) {
  1060→  const [stats, setStats] = useState<StatsDTO | null>(null);
  1061→  const [featured, setFeatured] = useState<ProjectListItem[] | null>(null);
  1062→  const [recent, setRecent] = useState<ProjectListItem[] | null>(null);
  1063→  const [departments, setDepartments] = useState<DepartmentDTO[] | null>(null);
  1064→  const [events, setEvents] = useState<EventDTO[] | null>(null);
  1065→
  1066→  useEffect(() => {
  1067→    const ac = new AbortController();
  1068→    fetchStats(ac.signal)
  1069→      .then(setStats)
  1070→      .catch(() => {});
  1071→    fetchDepartments(ac.signal)
  1072→      .then(setDepartments)
  1073→      .catch(() => {});
  1074→    fetchEvents(ac.signal)
  1075→      .then(setEvents)
  1076→      .catch(() => {});
  1077→    fetchProjects({ featured: "1", limit: 5 }, ac.signal)
  1078→      .then(setFeatured)
  1079→      .catch(() => {});
  1080→    fetchProjects({ sort: "newest", limit: 6 }, ac.signal)
  1081→      .then(setRecent)
  1082→      .catch(() => {});
  1083→    return () => ac.abort();
  1084→  }, []);
  1085→
  1086→  return (
  1087→    <div>
  1088→      <Hero navigate={navigate} stats={stats} recent={recent} />
  1089→      <Marquee projects={recent} />
  1090→      <StatsBand stats={stats} />
  1091→      <Spotlight navigate={navigate} projects={featured} />
  1092→      <Departments navigate={navigate} departments={departments} />
  1093→      <Events events={events} />
  1094→      <Latest navigate={navigate} recent={recent} />
  1095→      <CtaPanel navigate={navigate} />
  1096→    </div>
  1097→  );
  1098→}
