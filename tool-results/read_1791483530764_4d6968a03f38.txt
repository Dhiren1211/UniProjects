   121→   121→function stepForMessage(msg: string): number {
   122→   122→  const m = msg.toLowerCase();
   123→   123→  if (/technolog/.test(m)) return 4;
   124→   124→  if (/(member|teammate|username)/.test(m)) return 3;
   125→   125→  if (/(department|category|semester|supervisor|program|academic|year)/.test(m)) return 2;
   126→   126→  if (/(url|resource|timeline|milestone|date|link)/.test(m)) return 5;
   127→   127→  return 1; // title / summary / description and anything generic
   128→   128→}

   184→   184→/** Debounced user search + per-row role select, shared by the team step. */
   185→   185→function UserSearchAdd({
   186→   186→  exclude,
   187→   187→  onPick,
   188→   188→}: {
   189→   189→  exclude: Set<string>;
   190→   190→  onPick: (u: SafeUserDTO, role: string) => void;
   191→   191→}) {
   192→   192→  const [query, setQuery] = useState("");
   193→   193→  const [debounced, setDebounced] = useState("");
   194→   194→  const [found, setFound] = useState<SafeUserDTO[] | null>(null);
   195→   195→  const [foundFor, setFoundFor] = useState("");
   196→   196→  const [roles, setRoles] = useState<Record<string, string>>({});
   197→   197→  const [busyName, setBusyName] = useState<string | null>(null);
   198→   198→
   199→   199→  useEffect(() => {
   200→   200→    const t = setTimeout(() => setDebounced(query.trim().toLowerCase()), 300);
   201→   201→    return () => clearTimeout(t);
   202→   202→  }, [query]);
   203→   203→
   204→   204→  useEffect(() => {
   205→   205→    const q = debounced;
   206→   206→    if (q.length < 2) return;
   207→   207→    const ac = new AbortController();
   208→   208→    searchUsers(q, ac.signal)
   209→   209→      .then((rows) => {
   210→   210→        setFound(rows);
   211→   211→        setFoundFor(q);
   212→   212→      })
   213→   213→      .catch(() => {});
   214→   214→    return () => ac.abort();
   215→   215→  }, [debounced]);
   216→   216→
   217→   217→  const ready = debounced.length >= 2 && foundFor === debounced;
   218→   218→  const visible = ready ? (found ?? []).filter((u) => !exclude.has(u.username.toLowerCase())) : null;
   219→   219→
   220→   220→  const pick = (u: SafeUserDTO) => {
   221→   221→    setBusyName(u.id);
   222→   222→    setQuery("");
   223→   223→    Promise.resolve(onPick(u, roles[u.id] ?? "MEMBER"))
   224→   224→      .catch(() => {})
   225→   225→      .finally(() => setBusyName(null));
   226→   226→  };
   227→   227→
   228→   228→  return (
   229→   229→    <div>
   230→   230→      <Label htmlFor="team-search">Find teammates</Label>
   231→   231→      <div className="relative mt-1.5">
   232→   232→        <Search
   233→   233→          className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   234→   234→          aria-hidden
   235→   235→        />
   236→   236→        <Input
   237→   237→          id="team-search"
   238→   238→          value={query}
   239→   239→          onChange={(e) => setQuery(e.target.value)}
   240→   240→          placeholder="Search students by name or username…"
   241→   241→          className="h-10 rounded-xl bg-white pl-9"
   242→   242→          autoComplete="off"
   243→   243→        />
   244→   244→      </div>
   245→   245→      <div className="mt-2 max-h-56 space-y-1.5 overflow-y-auto nice-scroll">
   246→   246→        {visible === null ? (
   247→   247→          <p className="px-1 py-2 text-xs text-muted-foreground">
   248→   248→            {debounced.length >= 2 ? "Searching…" : "Type at least 2 characters to search."}
   249→   249→          </p>
   250→   250→        ) : visible.length === 0 ? (
   251→   251→          <p className="px-1 py-2 text-xs text-muted-foreground">No matching students.</p>
   252→   252→        ) : (
   253→   253→          visible.map((u) => (
   254→   254→            <div
   255→   255→              key={u.id}
   256→   256→              className="flex items-center gap-2.5 rounded-xl border border-up-border bg-white px-3 py-2"
   257→   257→            >
   258→   258→              <MemberAvatar name={u.displayName} ring={false} className="size-7" />
   259→   259→              <div className="min-w-0 flex-1">
   260→   260→                <p className="truncate text-sm font-medium text-foreground">{u.displayName}</p>
   261→   261→                <p className="truncate text-xs text-muted-foreground">@{u.username}</p>
   262→   262→              </div>
   263→   263→              <Select value={roles[u.id] ?? "MEMBER"} onValueChange={(role) => setRoles((r) => ({ ...r, [u.id]: role }))}>
   264→   264→                <SelectTrigger
   265→   265→                  aria-label={`Role for ${u.displayName}`}
   266→   266→                  className="h-8 w-[92px] rounded-lg text-xs"
   267→   267→                >
   268→   268→                  <SelectValue />
   269→   269→                </SelectTrigger>
   270→   270→                <SelectContent>
   271→   271→                  <SelectItem value="LEAD">Lead</SelectItem>
   272→   272→                  <SelectItem value="MEMBER">Member</SelectItem>
   273→   273→                </SelectContent>
   274→   274→              </Select>
   275→   275→              <Button
   276→   276→                size="sm"
   277→   277→                className="h-8 rounded-lg px-3 font-semibold"
   278→   278→                disabled={busyName === u.id}
   279→   279→                onClick={() => pick(u)}
   280→   280→              >
   281→   281→                {busyName === u.id ? <Loader2 className="size-3.5 animate-spin" aria-hidden /> : "Add"}
   282→   282→              </Button>
   283→   283→            </div>
   284→   284→          ))
   285→   285→        )}
   286→   286→      </div>
   287→   287→    </div>
   288→   288→  );
   289→   289→}
