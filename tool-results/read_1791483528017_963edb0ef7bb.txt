   120→/** Maps a server error message back to the wizard step that most likely caused it. */
   121→function stepForMessage(msg: string): number {
   122→  const m = msg.toLowerCase();
   123→  if (/technolog/.test(m)) return 4;
   124→  if (/(member|teammate|username)/.test(m)) return 3;
   125→  if (/(department|category|semester|supervisor|program|academic|year)/.test(m)) return 2;
   126→  if (/(url|resource|timeline|milestone|date|link)/.test(m)) return 5;
   127→  return 1; // title / summary / description and anything generic
   128→}

   151→function Stepper({ current, onGo }: { current: number; onGo: (n: number) => void }) {
   152→  return (
   153→    <ol className="flex flex-wrap items-center gap-y-2" aria-label="Wizard steps">
   154→      {STEPS.map((s, i) => {
   155→        const done = s.n < current;
   156→        const isCurrent = s.n === current;
   157→        return (
   158→          <li key={s.n} className="flex items-center">
   159→            {i > 0 ? (
   160→              <span aria-hidden className="mx-1.5 h-px w-5 bg-up-border-strong md:mx-2 md:w-8" />
   161→            ) : null}
   162→            <button
   163→              type="button"
   164→              disabled={!done}
   165→              onClick={() => onGo(s.n)}
   166→              aria-current={isCurrent ? "step" : undefined}
   167→              className={cn(
   168→                "inline-flex h-9 items-center gap-1.5 rounded-full px-3.5 text-xs font-semibold transition-all duration-200 md:text-sm",
   169→                done && "cursor-pointer bg-primary-500 text-white hover:bg-primary-600",
   170→                isCurrent && "border-2 border-primary-500 bg-white text-primary-700",
   171→                !done && !isCurrent && "cursor-default bg-surface-muted text-muted-foreground"
   172→              )}
   173→            >
   174→              {done ? <Check className="size-3.5" aria-hidden /> : <span className="tabular-nums">{s.n}</span>}

   176→            </button>

   179→      })}

   184→/** Debounced user search + per-row role select, shared by the team step. */
   185→function UserSearchAdd({
   186→  exclude,
   187→  onPick,
   188→}: {
   189→  exclude: Set<string>;
   190→  onPick: (u: SafeUserDTO, role: string) => void;
   191→}) {
   192→  const [query, setQuery] = useState("");
   193→  const [debounced, setDebounced] = useState("");
   194→  const [found, setFound] = useState<SafeUserDTO[] | null>(null);
   195→  const [foundFor, setFoundFor] = useState("");
   196→  const [roles, setRoles] = useState<Record<string, string>>({});
   197→  const [busyName, setBusyName] = useState<string | null>(null);
   198→
   199→  useEffect(() => {
   200→    const t = setTimeout(() => setDebounced(query.trim().toLowerCase()), 300);
   201→    return () => clearTimeout(t);
   202→  }, [query]);
   203→
   204→  useEffect(() => {
   205→    const q = debounced;
   206→    if (q.length < 2) return;
   207→    const ac = new AbortController();
   208→    searchUsers(q, ac.signal)
   209→      .then((rows) => {
   210→        setFound(rows);
   211→        setFoundFor(q);
   212→      })
   213→      .catch(() => {});
   214→    return () => ac.abort();
   215→  }, [debounced]);
   216→
   217→  const ready = debounced.length >= 2 && foundFor === debounced;
   218→  const visible = ready ? (found ?? []).filter((u) => !exclude.has(u.username.toLowerCase())) : null;
   219→
   220→  const pick = (u: SafeUserDTO) => {
   221→    setBusyName(u.id);
   222→    setQuery("");
   223→    Promise.resolve(onPick(u, roles[u.id] ?? "MEMBER"))
   224→      .catch(() => {})
   225→      .finally(() => setBusyName(null));
   226→  };
   227→
   228→  return (
   229→    <div>
   230→      <Label htmlFor="team-search">Find teammates</Label>
   231→      <div className="relative mt-1.5">
   232→        <Search
   233→          className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   234→          aria-hidden
   235→        />
   236→        <Input
   237→          id="team-search"
   238→          value={query}
   239→          onChange={(e) => setQuery(e.target.value)}
   240→          placeholder="Search students by name or username…"
   241→          className="h-10 rounded-xl bg-white pl-9"
   242→          autoComplete="off"
   243→        />
   244→      </div>
   245→      <div className="mt-2 max-h-56 space-y-1.5 overflow-y-auto nice-scroll">
   246→        {visible === null ? (
   247→          <p className="px-1 py-2 text-xs text-muted-foreground">
   248→            {debounced.length >= 2 ? "Searching…" : "Type at least 2 characters to search."}
   249→          </p>
   250→        ) : visible.length === 0 ? (
   251→          <p className="px-1 py-2 text-xs text-muted-foreground">No matching students.</p>
   252→        ) : (
   253→          visible.map((u) => (
   254→            <div
   255→              key={u.id}
   256→              className="flex items-center gap-2.5 rounded-xl border border-up-border bg-white px-3 py-2"
   257→            >
   258→              <MemberAvatar name={u.displayName} ring={false} className="size-7" />
   259→              <div className="min-w-0 flex-1">
   260→                <p className="truncate text-sm font-medium text-foreground">{u.displayName}</p>
   261→                <p className="truncate text-xs text-muted-foreground">@{u.username}</p>
   262→              </div>
   263→              <Select value={roles[u.id] ?? "MEMBER"} onValueChange={(role) => setRoles((r) => ({ ...r, [u.id]: role }))}>
   264→                <SelectTrigger
   265→                  aria-label={`Role for ${u.displayName}`}
   266→                  className="h-8 w-[92px] rounded-lg text-xs"
   267→                >
   268→                  <SelectValue />
   269→                </SelectTrigger>
   270→                <SelectContent>
   271→                  <SelectItem value="LEAD">Lead</SelectItem>
   272→                  <SelectItem value="MEMBER">Member</SelectItem>
   273→                </SelectContent>
   274→              </Select>
   275→              <Button
   276→                size="sm"
   277→                className="h-8 rounded-lg px-3 font-semibold"
   278→                disabled={busyName === u.id}
   279→                onClick={() => pick(u)}
   280→              >
   281→                {busyName === u.id ? <Loader2 className="size-3.5 animate-spin" aria-hidden /> : "Add"}
   282→              </Button>
   283→            </div>
   284→          ))
   285→        )}
   286→      </div>
   287→    </div>
   288→  );
   289→}
