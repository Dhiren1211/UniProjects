   186→/** "/project/<slug>" → internal view, anything else is not navigable. */
   187→function activityToView(href: string): View | null {
   188→  const project = /^\/project\/([^/?#]+)/.exec(href);
   189→  if (project) return { name: "project", slug: project[1] };
   190→  return null;
   191→}

   193→/** "2025-11" → "Nov". */
   194→function monthLabel(key: string) {
   195→  const [year, month] = key.split("-");
   196→  const idx = Number(month) - 1;
   197→  if (!year || !Number.isFinite(idx) || idx < 0 || idx > 11) return key;
   198→  return format(new Date(Number(year), idx, 1), "MMM");
   199→}

   201→/** value → debounced value (300 ms) — keeps search/filters off the network until typing settles. */
   202→function useDebounced<T>(value: T, delay = 300): T {
   203→  const [debounced, setDebounced] = useState(value);
   204→  useEffect(() => {
   205→    const t = setTimeout(() => setDebounced(value), delay);
   206→    return () => clearTimeout(t);
   207→  }, [value, delay]);
   208→  return debounced;
   209→}

   215→export function AdminView({ navigate }: { navigate: Navigate }) {
   216→  const { user, authLoaded, openAuth } = useAppStore();
   217→  const reduceMotion = useReducedMotion();
   218→  const [tab, setTab] = useState<TabKey>("overview");
   219→  const [overview, setOverview] = useState<AdminOverviewDTO | null>(null);
   220→  const [overviewError, setOverviewError] = useState(false);
   221→  const [refreshTick, setRefreshTick] = useState(0);
   222→  const [refreshing, setRefreshing] = useState(false);
   223→  const [loadedAt, setLoadedAt] = useState<Partial<Record<TabKey, number>>>({});
   224→  const [arriveNew, setArriveNew] = useState(false);
   225→
   226→  const isAdmin = user?.role === "ADMIN";
   227→
   228→  // Keep the "Updated …" captions honest without refetching anything.
   229→  const [, setClock] = useState(0);
   230→  useEffect(() => {
   231→    const t = setInterval(() => setClock((n) => n + 1), 30_000);
   232→    return () => clearInterval(t);
   233→  }, []);
   234→
   235→  /** Record a finished load: always stops the spinner; timestamps only on success. */
   236→  const noteLoaded = useCallback((keys: TabKey[], ok: boolean) => {
   237→    setRefreshing(false);
   238→    if (!ok) return;
   239→    const at = Date.now();
   240→    setLoadedAt((prev) => {
   241→      const next = { ...prev };
   242→      for (const key of keys) next[key] = at;
   243→      return next;
   244→    });
   245→  }, []);
   246→
   247→  // Stable per-tab onLoaded callbacks — tabs include them in their fetch-effect
   248→  // deps, so a fresh identity per render would trigger a refetch loop.
   249→  const overviewLoaded = useCallback(
   250→    (ok: boolean) => noteLoaded(["overview", "moderation"], ok),
   251→    [noteLoaded]
   252→  );
   253→  const projectsLoaded = useCallback((ok: boolean) => noteLoaded(["projects"], ok), [noteLoaded]);
   254→  const usersLoaded = useCallback((ok: boolean) => noteLoaded(["users"], ok), [noteLoaded]);
   255→  const departmentsLoaded = useCallback(
   256→    (ok: boolean) => noteLoaded(["departments"], ok),
   257→    [noteLoaded]
   258→  );
   259→
   260→  /** Explicit refresh from the content header — spins the button until data lands. */
   261→  const refresh = () => {
   262→    setRefreshing(true);
   263→    setRefreshTick((n) => n + 1);
   264→  };
   265→
   266→  /** Background re-sync of shared overview data after a mutation (no spinner). */
   267→  const syncOverview = () => setRefreshTick((n) => n + 1);
   268→
   269→  /** Switch tabs; quick actions can request the tab's "New …" dialog to open on arrival. */
   270→  const goTab = (key: TabKey, openNew = false) => {
   271→    setTab(key);
   272→    setArriveNew(openNew);
   273→    window.scrollTo({ top: 0 });
   274→  };
   275→
   276→  useEffect(() => {
   277→    if (!authLoaded || !isAdmin) return;
   278→    let cancelled = false;
   279→    fetchAdminOverview()
   280→      .then((data) => {
   281→        if (cancelled) return;
   282→        setOverview(data);
   283→        setOverviewError(false);
   284→        overviewLoaded(true);
   285→      })
   286→      .catch((e) => {
   287→        if (cancelled) return;
   288→        setOverviewError(true);
   289→        overviewLoaded(false);
   290→        toast.error(e instanceof Error ? e.message : "Could not load the admin overview");
   291→      });
   292→    return () => {
   293→      cancelled = true;
   294→    };
   295→  }, [refreshTick, authLoaded, isAdmin, overviewLoaded]);
   296→
   297→  // ---------- gates ----------
   298→
   299→  if (!authLoaded) {
   300→    return (
   301→      <Shell>
   302→        <div className="space-y-6">
   303→          <Skeleton className="h-24 w-full rounded-2xl" />
   304→          <Skeleton className="h-14 w-full rounded-2xl" />
   305→          <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   306→            {Array.from({ length: 8 }).map((_, i) => (
   307→              <Skeleton key={i} className="h-28 w-full rounded-2xl" />
   308→            ))}
   309→          </div>
   310→        </div>
   311→      </Shell>
   312→    );
   313→  }
   314→
   315→  if (!user || user.role !== "ADMIN") {
   316→    return (
   317→      <Shell>
   318→        <EmptyState
   319→          icon={user ? ShieldAlert : Lock}
   320→          title="Admin access required"
   321→          hint="This console manages the whole showcase — projects, members, departments and events. It is limited to Kyungsung University administrators."
   322→          action={
   323→            !user ? (
   324→              <Button onClick={() => openAuth("login")} className="rounded-xl font-semibold">
   325→                Sign in
   326→              </Button>
   327→            ) : undefined
   328→          }
   329→        />
   330→      </Shell>
   331→    );
   332→  }
   333→
   334→  const moderationPending = (overview?.pendingVerifications ?? 0) + (overview?.openReports ?? 0);
   335→  const meta = TAB_META[tab];
   336→
   337→  return (
   338→    <div className="lg:flex">
   339→      <ConsoleSidebar
   340→        tab={tab}
   341→        moderationPending={moderationPending}
   342→        onTab={goTab}
   343→        navigate={navigate}
   344→      />
   345→
   346→      <main className="min-w-0 flex-1 px-4 py-5 md:px-6 md:py-6">
   347→          {/* < lg: brand + scrollable pills replace the sidebar */}
   348→          <ConsoleTopBar
   349→            tab={tab}
   350→            moderationPending={moderationPending}
   351→            user={user}
   352→            onTab={goTab}
   353→          />
   354→
   355→          {/* < lg: the sidebar system mini-card lives at the top of Overview */}
   356→          {tab === "overview" && overview ? (
   357→            <SystemMiniCard storage={overview.storage} className="mt-4 lg:hidden" />
   358→          ) : null}
   359→
   360→          <ContentHeader
   361→            title={meta.title}
   362→            description={meta.description}
   363→            loadedAt={loadedAt[tab]}
   364→            refreshing={refreshing}
   365→            onRefresh={refresh}
   366→            className="mt-5"
   367→          />
   368→
   369→          {/* ---------- active tab ---------- */}
   370→          <div className="mt-4">
   371→            <AnimatePresence mode="wait" initial={false}>
   372→              <motion.div
   373→                key={tab}
   374→                initial={reduceMotion ? false : { opacity: 0, y: 10 }}
   375→                animate={{ opacity: 1, y: 0 }}
   376→                exit={reduceMotion ? { opacity: 0 } : { opacity: 0, y: -8 }}
   377→                transition={{ duration: 0.18, ease: "easeOut" }}
   378→              >
   379→            {tab === "overview" ? (
   380→              <OverviewTab
   381→                overview={overview}
   382→                error={overviewError}
   383→                onRetry={refresh}
   384→                navigate={navigate}
   385→                onQuickAction={goTab}
   386→              />
   387→            ) : null}
   388→            {tab === "projects" ? (
   389→              <ProjectsTab
   390→                navigate={navigate}
   391→                onChanged={syncOverview}
   392→                refreshTick={refreshTick}
   393→                onLoaded={projectsLoaded}
   394→              />
   395→            ) : null}
   396→            {tab === "users" ? (
   397→              <UsersTab
   398→                onChanged={syncOverview}
   399→                refreshTick={refreshTick}
   400→                onLoaded={usersLoaded}
   401→                autoOpenNew={arriveNew}
   402→              />
   403→            ) : null}
   404→            {tab === "departments" ? (
   405→              <DepartmentsTab
   406→                onChanged={syncOverview}
   407→                refreshTick={refreshTick}
   408→                onLoaded={departmentsLoaded}
   409→                autoOpenNew={arriveNew}
   410→              />
   411→            ) : null}
   412→            {tab === "moderation" ? <ModerationTab overview={overview} /> : null}
   413→              </motion.div>
   414→            </AnimatePresence>
   415→          </div>
   416→      </main>
   417→    </div>
   418→  );
   419→}

   517→    </aside>
   518→  );

   664→/**
   665→ * Fixed-size page window — the pager always renders the same slots:
   666→ *   edges:  `1 2 3 4 5 … N`  /  `1 … N-4 N-3 N-2 N-1 N`
   667→ *   middle: `1 … p-1 p p+1 … N`
   668→ * so its width never changes while navigating (e.g. `1 2 3 … 100`).
   669→ */
   670→function pageList(page: number, totalPages: number): (number | "…")[] {
   671→  if (totalPages <= 5) return Array.from({ length: totalPages }, (_, i) => i + 1);
   672→  if (page <= 3) return [1, 2, 3, 4, 5, "…", totalPages];
   673→  if (page >= totalPages - 2)
   674→    return [1, "…", totalPages - 4, totalPages - 3, totalPages - 2, totalPages - 1, totalPages];
   675→  return [1, "…", page - 1, page, page + 1, "…", totalPages];
   676→}

   678→/** Minimal pager: “Showing a–b of N” left, compact page buttons right. Hides itself when it fits on one page. */
   679→function Pagination({
   680→  page,

   698→    "inline-flex h-8 min-w-8 items-center justify-center rounded-lg px-1.5 text-xs font-semibold tabular-nums transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring";

   705→        Showing <span className="font-semibold text-foreground">{from}–{to}</span> of{" "}

   715→              page === 1 && "pointer-events-none opacity-40"
   716→            )}
   717→            onClick={() => onPage(page - 1)}
   718→            disabled={page === 1}
   719→            aria-label="Previous page"
   720→          >
   721→            <ChevronLeft className="size-4" aria-hidden />

   751→            className={cn(
   752→              btn,
   753→              "text-muted-foreground hover:bg-surface-muted hover:text-foreground",
   754→              page === totalPages && "pointer-events-none opacity-40"
   755→            )}
   756→            onClick={() => onPage(page + 1)}
   757→            disabled={page === totalPages}
   758→            aria-label="Next page"
   759→          >
   760→            <ChevronRight className="size-4" aria-hidden />
   761→          </button>
   762→        </div>

   866→/** rAF count-up that respects reduced motion — gives every number a live "landing" feel. */
   867→function useCountUp(target: number, duration = 900): number {
   868→  const reduceMotion = useReducedMotion();
   869→  const [value, setValue] = useState(0);
   870→  useEffect(() => {
   871→    if (reduceMotion) return; // derived below — no animation needed
   872→    let raf = 0;
   873→    const start = performance.now();
   874→    const tick = (now: number) => {
   875→      const t = Math.min(1, (now - start) / duration);
   876→      const eased = 1 - Math.pow(1 - t, 3); // ease-out cubic
   877→      setValue(Math.round(target * eased));
   878→      if (t < 1) raf = requestAnimationFrame(tick);
   879→    };
   880→    raf = requestAnimationFrame(tick);
   881→    return () => cancelAnimationFrame(raf);
   882→  }, [target, duration, reduceMotion]);
   883→  return reduceMotion ? target : value;
   884→}

  1514→/** "Community make-up" — SVG donut of members by role with center total + legend. */

  1520→  className?: string;
  1521→}) {
