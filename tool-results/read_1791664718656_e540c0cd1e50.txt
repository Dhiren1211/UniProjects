   892→/** rAF count-up that respects reduced motion — gives every number a live "landing" feel. */
   893→function useCountUp(target: number, duration = 900): number {
   894→  const reduceMotion = useReducedMotion();
   895→  const [value, setValue] = useState(0);
   896→  useEffect(() => {
   897→    if (reduceMotion) return; // derived below — no animation needed
   898→    let raf = 0;
   899→    const start = performance.now();
   900→    const tick = (now: number) => {
   901→      const t = Math.min(1, (now - start) / duration);
   902→      const eased = 1 - Math.pow(1 - t, 3); // ease-out cubic
   903→      setValue(Math.round(target * eased));
   904→      if (t < 1) raf = requestAnimationFrame(tick);
   905→    };
   906→    raf = requestAnimationFrame(tick);
   907→    return () => cancelAnimationFrame(raf);
   908→  }, [target, duration, reduceMotion]);
   909→  return reduceMotion ? target : value;
   910→}
