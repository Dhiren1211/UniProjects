   133→   892→/** rAF count-up that respects reduced motion — gives every number a live "landing" feel. */
   134→   893→function useCountUp(target: number, duration = 900): number {
   135→   894→  const reduceMotion = useReducedMotion();
   136→   895→  const [value, setValue] = useState(0);
   137→   896→  useEffect(() => {
   138→   897→    if (reduceMotion) return; // derived below — no animation needed
   139→   898→    let raf = 0;
   140→   899→    const start = performance.now();
   141→   900→    const tick = (now: number) => {
   142→   901→      const t = Math.min(1, (now - start) / duration);
   143→   902→      const eased = 1 - Math.pow(1 - t, 3); // ease-out cubic
   144→   903→      setValue(Math.round(target * eased));
   145→   904→      if (t < 1) raf = requestAnimationFrame(tick);
   146→   905→    };
   147→   906→    raf = requestAnimationFrame(tick);
   148→   907→    return () => cancelAnimationFrame(raf);
   149→   908→  }, [target, duration, reduceMotion]);
   150→   909→  return reduceMotion ? target : value;
   151→   910→}
